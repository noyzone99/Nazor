cmake_minimum_required(VERSION 3.20)
project(nazor_native VERSION 1.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64|aarch64|arm64|ARM64)$" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Nazor native assets require a 64-bit x64 or ARM64 Linux target.")
endif()
include(CTest)
find_package(PkgConfig REQUIRED)
option(NAZOR_WAYLAND "Build the Wayland bridge" ON)
option(NAZOR_X11 "Build the X11 bridge" ON)
option(NAZOR_X11_WINDOW_TESTS "Run the isolated X11 server tests" ON)
if(NAZOR_WAYLAND)
find_package(Threads REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client>=1.20 wayland-cursor)
pkg_check_modules(XKB REQUIRED IMPORTED_TARGET xkbcommon)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
set(PROTOCOL_SOURCES)
file(GLOB PROTOCOLS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Protocols/*.xml")
foreach(XML IN LISTS PROTOCOLS)
  get_filename_component(NAME "${XML}" NAME_WE)
  set(HEADER "${CMAKE_CURRENT_BINARY_DIR}/${NAME}-client-protocol.h")
  set(CODE "${CMAKE_CURRENT_BINARY_DIR}/${NAME}-protocol.c")
  add_custom_command(OUTPUT "${HEADER}" COMMAND "${WAYLAND_SCANNER}" client-header "${XML}" "${HEADER}" DEPENDS "${XML}")
  add_custom_command(OUTPUT "${CODE}" COMMAND "${WAYLAND_SCANNER}" private-code "${XML}" "${CODE}" DEPENDS "${XML}")
  list(APPEND PROTOCOL_SOURCES "${HEADER}" "${CODE}")
endforeach()
add_library(nazor-wayland-protocols OBJECT ${PROTOCOL_SOURCES})
set_target_properties(nazor-wayland-protocols PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(nazor-wayland-protocols PRIVATE -fvisibility=hidden)
target_link_libraries(nazor-wayland-protocols PRIVATE PkgConfig::WAYLAND)
add_library(nazor-wayland SHARED nazor-wayland.c)
target_compile_features(nazor-wayland PRIVATE c_std_11)
target_compile_options(nazor-wayland PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter -fvisibility=hidden)
target_include_directories(nazor-wayland PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(nazor-wayland PRIVATE nazor-wayland-protocols PkgConfig::WAYLAND PkgConfig::XKB Threads::Threads m)
if(BUILD_TESTING)
  add_executable(nazor-wayland-tests tests.c)
  target_link_libraries(nazor-wayland-tests PRIVATE nazor-wayland)
  add_test(NAME bridge COMMAND nazor-wayland-tests)
  add_executable(nazor-wayland-input-tests input-tests.c)
  target_compile_features(nazor-wayland-input-tests PRIVATE c_std_11)
  target_compile_options(nazor-wayland-input-tests PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
  target_include_directories(nazor-wayland-input-tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(nazor-wayland-input-tests PRIVATE nazor-wayland-protocols PkgConfig::WAYLAND PkgConfig::XKB Threads::Threads m)
  add_test(NAME input COMMAND nazor-wayland-input-tests)
endif()
endif()
if(NAZOR_X11)
  pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11 xcursor xrender xext xrandr)
  find_package(Threads REQUIRED)
  add_library(nazor-x11 SHARED nazor-x11.c)
  target_compile_features(nazor-x11 PRIVATE c_std_11)
  target_compile_options(nazor-x11 PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter -fvisibility=hidden)
  target_link_libraries(nazor-x11 PRIVATE PkgConfig::X11 Threads::Threads m)
  target_link_options(nazor-x11 PRIVATE -Wl,-z,nodelete)
  if(BUILD_TESTING)
    add_executable(nazor-x11-tests x11-tests.c)
    target_link_libraries(nazor-x11-tests PRIVATE nazor-x11 PkgConfig::X11 Threads::Threads)
    add_test(NAME x11-no-display COMMAND nazor-x11-tests --no-display)
    find_program(XVFB Xvfb)
    find_program(OPENBOX openbox)
    find_program(XPROP xprop)
    if(XVFB AND OPENBOX AND XPROP AND NAZOR_X11_WINDOW_TESTS)
      add_test(NAME x11-windows COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/x11-test-server.sh" "$<TARGET_FILE:nazor-x11-tests>")
    endif()
    find_program(WESTON weston)
    find_program(XWAYLAND Xwayland)
    find_library(XTEST_LIBRARY NAMES Xtst libXtst.so.6)
    if(XVFB AND WESTON AND XWAYLAND AND XTEST_LIBRARY AND NAZOR_X11_WINDOW_TESTS)
      target_compile_definitions(nazor-x11-tests PRIVATE NAZOR_X11_NESTED_TESTS)
      target_link_libraries(nazor-x11-tests PRIVATE "${XTEST_LIBRARY}")
      add_test(NAME xwayland-outer-resize COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/xwayland-test-server.sh"
        "${WESTON}" "${XWAYLAND}" "$<TARGET_FILE:nazor-x11-tests>")
      set_tests_properties(xwayland-outer-resize PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
    endif()
  endif()
endif()
